void blinkLED(){
  watchcounter++;
  //Serial.print(watchcounter);
  if(watchcounter > 20){  
    //ets_printf("reboot\n");   
    //Serial.println("reboot."); 
    WiFi.persistent(false); 
    WiFi.disconnect();
    WiFi.mode(WIFI_OFF);
    WiFi.mode(WIFI_STA);
    ESP.restart();
  }


    
}
void startWatchdog(){    
    //Initialize Ticker every 0.5s
    blinker.attach(2, blinkLED); // use attach_ms for milliseconds
}

void connect() {
  Serial.print("checking wifi\n");  
  while (WiFi.status() != WL_CONNECTED) {    
    if (lastwifibegin < millis() + 30000) {
      WiFi.persistent(false); 
      WiFi.disconnect();
      WiFi.mode(WIFI_OFF);
      WiFi.mode(WIFI_STA);
      lastwifibegin = millis();
      Serial.print("\nretry wifi begin...");
      Serial.println(millis());
      //WiFi.config(INADDR_NONE, INADDR_NONE, INADDR_NONE);
      WiFi.hostname(hostname);
      WiFi.begin(ssid, password);
      while (WiFi.status() != WL_CONNECTED) {  
        Serial.print(".");
        delay(1000); //warten auf Verbindung  
      }
      Serial.println("\n-------------------------");      
      Serial.print("IP address: ");
      Serial.println(WiFi.localIP());
      Serial.print("ESP Mac Address: ");
      Serial.println(WiFi.macAddress());
      Serial.print("Subnet Mask: ");
      Serial.println(WiFi.subnetMask());
      Serial.print("Gateway IP: ");
      Serial.println(WiFi.gatewayIP());
      Serial.print("DNS: ");
      Serial.println(WiFi.dnsIP());  
      Serial.println("-------------------------");      
      Serial.print("\nwifi connected.\n");
      for (int i=0; i<3; i++){
        digitalWrite(ledPin, 1); // Status   
        digitalWrite(relPin, 1); // Status   
        delay(100);
        digitalWrite(ledPin, 0); // Status   
        digitalWrite(relPin, 0); // Status   
        delay(100);
      }
      break;      
    }
  }
  
  //printMacAddress();

  Serial.print("connecting to MQTT-Server");
  for (int i=0; i<3; i++){
    Serial.print("\nconnecting to MQTT-Server");
    digitalWrite(ledPin, 1); // Status   
    
    if(mclient.connect(mqtt_user, "try", "try")){
      Serial.println("\nMQTT connected.\n");
      break;     
    }
    Serial.print("not connecting to MQTT-Server");
    digitalWrite(ledPin, 0); // Status   
    delay(1000);
  }
  //endeZeit = millis();  
  mclient.subscribe(mqtt_path_receive);
  Serial.print("subscribe to ");
  Serial.print(mqtt_path_receive);
  Serial.print("\n");
  Serial.print("Connection complete.\n");
  delay(500);  
  digitalWrite(ledPin, 0); // Status   
  delay(1000);  
  digitalWrite(ledPin, 1); // Status   
  digitalWrite(relPin, 1);
}

void relChangeState()
{
  digitalWrite(relPin, !(digitalRead(relPin)));  //Invert Current State of LED  
}

{

//=======================================================================
//                MAIN LOOP
//=======================================================================
void loop()
{
  watchcounter = 0;
  mclient.loop();
  // blinken ohne unterbrechung des loop
  if(endeZeit != 0){    // wird gestartet von mqtt signal, blinkzeit gesamt    
    if( millis() < endeZeit ){ //blinken läuft          
        if(ledausZeit < millis()) { 
          if(digitalRead(ledPin)==0) {
            digitalWrite(ledPin, 1); // led aus   
            relChangeState(); //rel klick doppelt schnell
            ledausZeit = millis() + 500;
          } else {
            digitalWrite(ledPin, 0); // led ein   
            ledausZeit = millis() + 20;            
          }
        }   
      } else {      
      endeZeit=0;      
      digitalWrite(ledPin, 1); // led aus
      digitalWrite(relPin, 1); // led aus
      }
  }  
  if (!mclient.connected()) {
    connect();
  }

  //while (true){     Serial.println("waiting");    delay(1000);  }
}
